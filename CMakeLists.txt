cmake_minimum_required(VERSION 3.20)
project(PostcardCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_library(postcard_core STATIC core/postcard.cpp)
target_include_directories(postcard_core PUBLIC core)
add_executable(core_tests tests/core_test.cpp)
target_link_libraries(core_tests PRIVATE postcard_core)
enable_testing()
add_test(NAME core_tests COMMAND core_tests)
